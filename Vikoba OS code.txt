function approveAction(btn, status) {
    const card = btn.closest('.bg-white');
    const btns = card.querySelectorAll('button');
    btns.forEach(b => b.remove());
    const badge = document.createElement('span');
    badge.className = status === 'Imeidhinishwa' ? 'bg-emerald-50 text-emerald-700 text-xs px-3 py-1 rounded-full font-medium' : 'bg-red-50 text-red-600 text-xs px-3 py-1 rounded-full font-medium';
    badge.textContent = status;
    card.querySelector('div').appendChild(badge);
}

function processRepayment() {
    const mid = document.getElementById('repay-member').value;
    const amount = parseInt(document.getElementById('repay-amount').value);
    const member = members.find(m=>m.id===mid);
    if(member) member.loan = Math.max(0, member.loan - amount);
    const txId = genTxId();
    closeModal();
    openModal(`<div class="text-center">
        <div class="w-14 h-14 bg-blue-100 rounded-full flex items-center justify-center mx-auto mb-3"><i data-lucide="check" style="width:28px;height:28px" class="text-blue-600"></i></div>

            <p><span class="text-gray-500">TXN:</span> <span class="font-mono">${txId}</span></p>
            <p><span class="text-gray-500">Mwanachama:</span> ${member?member.name:''}</p>
            <p><span class="text-gray-500">Kiasi:</span> ${fmt(amount)}</p>
            <p><span class="text-gray-500">Baki Mpya:</span> ${member?fmt(member.loan):''}</p>

}
